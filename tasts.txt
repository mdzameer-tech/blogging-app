////hash + salt + pepper,

userSchema.pre("save", function (next){

    const user = this;
    if (!user.isModified("password")) return next();

    const salt = randomBytes(16).toString();
    const hashedPassword = createHmac("sha256", salt)
        .update(user.password)
        .digest("hex");

    user.salt = salt;
    user.password = hashedPassword;

});

////virtual fxn using static in mongoose,



userSchema.static("matchPassword", async function(email, password){
    const user = await this.findOne({ email });
    if(!user) throw new Error("user not found");

    const salt = user.salt;
    const hashedPassword = user.password;

    const providedPasswordHash = 
    createHmac("sha256", salt)
        .update(password)
        .digest("hex");
    
    if(hashedPassword !== providedPasswordHash){
        throw new Error("incorrect password")
    }


    return user;
})




const User = model("user", userSchema);

module.exports = User;

////also basic sign up logic with some errors handling 


router.post("/signup", async (req, res) => {
    const { fullName, email, password } = req.body;

    try {
        await User.create({ fullName, email, password });
        return res.redirect("/signin");
    } catch (err) {
        if (err.code === 11000) {
            // duplicate email
            return res.status(409).render("signup.ejs", {
                error: "An account with this email already exists."
            });
        }
        console.error(err);
        return res.status(500).render("signup.ejs", {
            error: "Something went wrong. Please try again."
        });
    }
});
